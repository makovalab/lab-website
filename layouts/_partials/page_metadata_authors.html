{{/* Author list for a publication. Overrides the theme, which links every author it can find a term page for. `author` is a taxonomy, so Hugo mints a page for each distinct name string across the publications -- 1,242 of them, nearly all people with no connection to the lab, each getting an author page on this site that says nothing. So a name is linked only when data/author_links.yaml says it belongs to a lab member or collaborator. Of those, the lab's own people -- the Members and Former Members groups in data/authors -- are bold, so lab authorship can be picked out of a long author list; collaborators are linked but not bold, and everyone else is plain text. That file is generated by scripts/author_links.py; a spelling it cannot work out -- a change of name, most of all -- goes in data/author_names_extra.yaml. */}} {{- $lab_groups := slice "Members" "Former Members" -}} {{- $authors := .Params.authors -}} {{ if $authors }} {{- if not (reflect.IsSlice $authors) -}} {{- $authors = slice $authors -}} {{- end -}} {{ range $index, $raw := $authors }} {{- $slug := index hugo.Data.author_links $raw -}} {{- $termPage := "" -}} {{- with $slug }}{{ $termPage = site.GetPage (printf "/authors/%s" .) }}{{ end -}} {{/* The name is printed as the paper published it -- "L Smeds", not the profile's "Linnéa Smeds". A citation should match its source, and the old profile lookup expanded only the names that happened to urlize to a slug, so one author line could mix both forms. */}} {{- $display := $raw -}} {{/* Groups come from the linked slug, not the urlized name: several author strings ("A Zaidi") do not urlize to the person's slug (aa-zaidi). */}} {{- $is_lab := false -}} {{- with $slug -}} {{- $linked := partial "functions/get_author_profile" (dict "slug" .) -}} {{- range $linked.user_groups -}} {{- if in $lab_groups . }}{{ $is_lab = true }}{{ end -}} {{- end -}} {{- end -}} {{- if gt $index 0 }}, {{ end -}} {{- if $termPage -}} {{ $display }} {{- else -}} {{ $display }} {{- end -}} {{- if isset $.Params "author_notes" -}} {{- with (index $.Params.author_notes $index) -}} {{- end -}} {{- end -}} {{ end }} {{ end }}
