{{/* Author list for a publication. Overrides the theme, which links every author it can find a term page for. `author` is a taxonomy, so Hugo mints a page for each distinct name string across the publications -- 1,242 of them, nearly all people with no connection to the lab, each getting an author page on this site that says nothing. So a name is linked only when data/author_links.yaml says it belongs to a lab member or collaborator, and lab names are shown in bold so they can be picked out of a long author list. Everyone else is plain text. That file is generated by scripts/author_links.py; a spelling it cannot work out -- a change of name, most of all -- goes in data/author_names_extra.yaml. */}} {{- $authors := .Params.authors -}} {{ if $authors }} {{- if not (reflect.IsSlice $authors) -}} {{- $authors = slice $authors -}} {{- end -}} {{ range $index, $raw := $authors }} {{- $slug := index hugo.Data.author_links $raw -}} {{- $termPage := "" -}} {{- with $slug }}{{ $termPage = site.GetPage (printf "/authors/%s" .) }}{{ end -}} {{- $profile := partial "functions/get_author_profile" (dict "slug" ($raw | urlize) "title" $raw) -}} {{- $display := $profile.title | default $raw -}} {{- if gt $index 0 }}, {{ end -}} {{- if $termPage -}} {{ $display }} {{- else -}} {{ $display }} {{- end -}} {{- if isset $.Params "author_notes" -}} {{- with (index $.Params.author_notes $index) -}} {{- end -}} {{- end -}} {{ end }} {{ end }}
